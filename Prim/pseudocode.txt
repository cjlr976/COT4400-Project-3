algorithm prim(Graph, startNode):
    visitedNodes = empty set
    mstEdges = empty list
    availableEdges = new PriorityQueue()
    totalWeight = 0

    visitedNodes.add(startNode)

    for each (neighbor, weight) connected to startNode:
        availableEdges.insert(weight, startNode, neighbor)

    while availableEdges is not empty:
        weight, sourceNode, targetNode = availableEdges.extractMin()

        if targetNode is not in visitedNodes:
            visitedNodes.add(targetNode)
            mstEdges.append(sourceNode, targetNode, weight)
            totalWeight = totalWeight + weight

        for each (nextNeighbor, nextWeight) connected to targetNode:
            if nextNeighbor is NOT in visitedNodes:
                availableEdges.insert(nextWeight, targetNode, nextNeighbor)

    return mstEdges, totalWeight
