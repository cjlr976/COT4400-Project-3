Algorithm bfs(graph, start):​

    queue = Deque()​
    queue.append(start)​

    visited = set()​
    visited.add(start)​

    order = []  ​
    while queue is not empty:​
        node = queue.popleft()​
        order.append(node)​

        for neighbor in graph[node]:​
            if neighbor not in visited:​
                visited.add(neighbor)​
                queue.append(neighbor)​

    return order
